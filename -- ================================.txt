-- Validate + Load
local unlocked = false
local function tryUnlock()
    if unlocked then return end
    local entered = (input.Text or ""):gsub("%s+", "")
    if entered == VALID_KEY then
        unlocked = true
        status.TextColor3 = Color3.fromRGB(80, 220, 120)
        status.Text = "Key correct! Loading hub..."
        submit.Text = "Loading..."
        submit.BackgroundColor3 = Color3.fromRGB(40, 160, 80)
        task.wait(0.5)
        pcall(function() gui:Destroy() end)
        local ok, err = pcall(function()
            loadstring(game:HttpGet(SCRIPT_URL))()
        end)
        if not ok then
            warn("[Rivo] Failed to load hub:", err)
        end
    else
        status.TextColor3 = Color3.fromRGB(255, 90, 90)
        status.Text = "Invalid key."
    end
end

submit.MouseButton1Click:Connect(tryUnlock)
input.FocusLost:Connect(function(enterPressed)
    if enterPressed then tryUnlock() end
end)
